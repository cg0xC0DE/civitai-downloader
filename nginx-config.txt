# Civitai Downloader nginx 配置片段
# 添加到现有 nginx.conf 的 server 块内（与其他 location 同级）
#
# 前端访问: http://localhost:8080/civitaidl/
# 后端API: http://localhost:8080/civitaidl-service/api/download
#
# 注意: 将 /path/to/civitai-downloader 替换为实际路径

# 前端: /civitaidl/ → 静态文件
location /civitaidl/ {
    root   C:/workplace/civitai-downloader;
    index  index.html;
}

# 后端: /civitaidl-service/ → API 代理
location /civitaidl-service/ {
    rewrite ^/civitaidl-service/(.*) /$1 break;
    proxy_pass http://127.0.0.1:53133/;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
}
